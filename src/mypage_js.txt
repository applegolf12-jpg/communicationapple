CA.ready.then(function(u){
 if(!CA.on){say('mp-off','지금은 로그인 기능을 준비 중입니다.');return}

   return '<tr><td>'+dt(a.created_at)+'</td><td><b>'+esc(a.place)+' '+a.session_no+'차</b><br><span style="color:#5A645F">'+esc(a.period)+(a.pro?' · '+esc(a.pro)+' 프로':'')+'</span></td><td>'+a.people+'명'+(a.single_rooms?'<br><span style="color:#5A645F">1인실 '+a.single_rooms+'</span>':'')+'</td><td>'+won(a.total_won)+'</td><td><span class="st" data-s="'+esc(a.status)+'">'+esc(a.status)+'</span><br><small style="color:#5A645F">'+(NOTE[a.status]||'')+'</small></td></tr>'}).join('')+'</tbody></table></div>'});

  if(r.error)return say('pf-msg',CA.err(r.error));CA.profile=r.data;$('pf-mk-at').textContent=r.data.marketing_agreed_at?'동의일 '+dt(r.data.marketing_agreed_at):'';say('pf-msg','저장했습니다.',true)})});

})();</script>
